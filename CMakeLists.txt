cmake_minimum_required(VERSION 3.20)
project(A64Sentinel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Avoid loading unrelated MinGW runtime DLLs from an earlier PATH entry.
if(MINGW)
  add_link_options(-static)
endif()
add_library(sentinel src/generator.cpp src/process.cpp src/testing.cpp src/io.cpp src/report.cpp src/minimize.cpp src/options.cpp)
target_include_directories(sentinel PUBLIC include)
add_executable(a64sentinel src/main.cpp)
target_link_libraries(a64sentinel PRIVATE sentinel)
if(MSVC)
  target_compile_options(sentinel PRIVATE /W4)
else()
  target_compile_options(sentinel PRIVATE -Wall -Wextra -Wpedantic)
endif()
