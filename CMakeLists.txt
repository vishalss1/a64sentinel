cmake_minimum_required(VERSION 3.20)
project(A64Sentinel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Avoid loading unrelated MinGW runtime DLLs from an earlier PATH entry.
if(MINGW)
  add_link_options(-static)
endif()
add_library(sentinel src/generator.cpp src/process.cpp src/testing.cpp src/io.cpp src/report.cpp src/minimize.cpp src/options.cpp)
target_include_directories(sentinel PUBLIC include)
add_executable(a64sentinel src/main.cpp)
target_link_libraries(a64sentinel PRIVATE sentinel)
if(MSVC)
  target_compile_options(sentinel PRIVATE /W4)
else()
  target_compile_options(sentinel PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
option(SENTINEL_FETCH_GTEST "Fetch GoogleTest if unavailable" ON)
if(BUILD_TESTING)
  find_package(GTest CONFIG QUIET)
  if(NOT GTest_FOUND AND SENTINEL_FETCH_GTEST)
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  if(TARGET GTest::gtest_main)
    add_executable(sentinel_tests tests/core_test.cpp)
    target_link_libraries(sentinel_tests PRIVATE sentinel GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(sentinel_tests)
  else()
    message(FATAL_ERROR "GoogleTest required: install GTest or enable SENTINEL_FETCH_GTEST; alternatively disable BUILD_TESTING")
  endif()
endif()
